## 6-1 既存システムとの連携

**RESTful API**:

```python
import requests

class LegacySystemConnector:
    def __init__(self, base_url):
        self.base_url = base_url
        self.session = requests.Session()
    
    def get_customer_data(self, customer_id):
        response = self.session.get(
            f"{self.base_url}/api/customers/{customer_id}"

        response = self.session.put(
            f"{self.base_url}/api/orders/{order_id}",

        )
```

### 統合パターン

#### ファサードパターン

```python
class LegacyFacade:
    def __init__(self, legacy_system):
        self.legacy = legacy_system
    
    def get_customer_info(self, customer_id):
        # レガシーシステムとの互換性変換
        legacy_data = self.legacy.query_customer(customer_id)
        return self._convert_to_modern(legacy_data)
    
    def _convert_to_modern(self, data):
        return {
            'id': data['cust_id'],
            'name': data['cust_name'],
            'email': data['email_addr']
        }
```

```python
class LegacyAdapter:
    def __init__(self, legacy_system):

    def process_order(self, order):

        legacy_format = self._to_legacy_format(order)
        result = self.legacy.submit_order(legacy_format)
```

#### JWT トークン検証

```python
import jwt
from functools import wraps

def require_auth(f):
    @wraps(f)
    def decorated(*args, **kwargs):
        token = request.headers.get('Authorization')
        
        if not token:
            return {'error': 'No token'}, 401
        
        try:
            payload = jwt.decode(
                token.split()[1],
                CONFIG['JWT_SECRET'],
                algorithms=['HS256']
            )
            request.user = payload
        except jwt.ExpiredSignatureError:
            return {'error': 'Token expired'}, 401
        except jwt.InvalidTokenError:
            return {'error': 'Invalid token'}, 401
        
        return f(*args, **kwargs)
    return decorated
```

## 6-3 データ同期と一貫性

### 同期パターン

#### 1. 完全同期

```python
def full_sync(source, target):
    # 全てのデータを取得
    all_data = source.get_all()
    
    # ターゲットに反映
    for item in all_data:
        target.upsert(item)
```

#### 2. 増分同期

```python
def incremental_sync(source, target, last_sync_time):
    # 変更されたデータのみ取得
    changed_data = source.get_modified_since(last_sync_time)
    
    # ターゲットに反映
    for item in changed_data:
        target.upsert(item)
    
    return datetime.now()
```

#### 3. CDC（Change Data Capture）

```python
from confluent_kafka import Consumer

def cdc_sync():
    consumer = Consumer({
        'bootstrap.servers': 'kafka:9092',
        'group.id': 'ai-sync-group',
        'auto.offset.reset': 'earliest'
    })
    
    consumer.subscribe(['database-changes'])
    
    while True:
        msg = consumer.poll(1.0)
        if msg is None:
            continue
        
        change = msg.value()
        apply_change(change)
```

### 一貫性モデル

#### 最終一貫性

```
Time →
System A: [Update] ────────▶
                    ↓
System B: ────────▶ [Update]
```

**実装**:
```python
class EventuallyConsistentStore:
    def __init__(self):
        self.queue = []
    
    def update(self, data):
        self.queue.append({
            'data': data,
            'timestamp': datetime.now()
        })
    
    def sync(self):
        while self.queue:
            change = self.queue.pop(0)
            self._apply(change)
```

#### 強い一貫性

```python
from redis import Redis

redis = Redis()

def strong_consistent_update(key, value):
    # トランザクション
    pipe = redis.pipeline()
    pipe.set(key, value)
    pipe.expire(key, 3600)
    pipe.execute()
```

#### デッドレターキュー

```python
class DeadLetterQueue:
    def __init__(self):
        self.failed_messages = []
    
    def add(self, message, error):
        self.failed_messages.append({
            'message': message,
            'error': str(error),
            'timestamp': datetime.now()
        })
    
    def retry_all(self):
        for item in self.failed_messages:
            try:
                process(item['message'])
            except:
                pass
```

### イベントパターン

#### Command-Query Responsibility Segregation (CQRS)

```
┌─────────┐     ┌──────────┐
│ Command │     │  Query   │
│  Write  │     │   Read   │
└────┬────┘     └────┬─────┘
     │               │
     ▼               ▼
┌─────────┐     ┌──────────┐
│ Write   │     │  Read    │
│ Model   │────▶│  Model   │
└─────────┘     └──────────┘
```

**実装**:

```python
class CommandHandler:
    def handle(self, command):
        # 状態変更
        self.repository.save(command.entity)
        
        # イベント発行

            EntityChanged(entity=command.entity)
        )

class QueryHandler:
    def handle(self, query):
```

#### Event Sourcing

```python
class EventSourcedEntity:
    def __init__(self):
        self.events = []
        self.state = self._initial_state()
    
    def apply_event(self, event):
        self.events.append(event)
        self._apply(event)
    
    def _apply(self, event):
        if isinstance(event, Created):
            self.state['id'] = event.id
        elif isinstance(event, Updated):
            self.state['value'] = event.value
    
    def get_state(self):
        return self.state
```
