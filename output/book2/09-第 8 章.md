```python
from collections import deque
import time

class ThroughputMonitor:
    def __init__(self, window_size=60):
        self.requests = deque(maxlen=window_size)
        self.window_size = window_size
    
    def record(self):
        self.requests.append(time.time())
    
    def get_rps(self):
        if not self.requests:
            return 0
        window = time.time() - self.requests[0]
        return len(self.requests) / window if window > 0 else 0
```

```python
class ErrorRateMonitor:

    def record_request(self):
```

### アラート設定

#### しきい値ベース

```python
class AlertManager:
    def __init__(self):
        self.thresholds = {
            'latency_p99': {'warning': 1.0, 'critical': 2.0},
            'error_rate': {'warning': 0.05, 'critical': 0.1},
            'throughput': {'warning': 10, 'critical': 5}
        }
    
    def check_alerts(self, metrics):
        alerts = []
        
        # レイテンシチェック
        if metrics['latency_p99'] > self.thresholds['latency_p99']['critical']:
            alerts.append({
                'level': 'critical',
                'metric': 'latency_p99',
                'value': metrics['latency_p99']
            })
        
        # エラーレートチェック
        if metrics['error_rate'] > self.thresholds['error_rate']['critical']:
            alerts.append({
                'level': 'critical',
                'metric': 'error_rate',
                'value': metrics['error_rate']
            })
        
        return alerts
```

### コスト追跡

#### API コスト監視

```python
class CostTracker:
    def __init__(self, rates):
        self.rates = rates  # $/1K tokens
        self.total_cost = 0
        self.usage = {
            'input_tokens': 0,
            'output_tokens': 0
        }
    
    def record_usage(self, input_tokens, output_tokens):
        self.usage['input_tokens'] += input_tokens
        self.usage['output_tokens'] += output_tokens
        
        input_cost = (input_tokens / 1000) * self.rates['input']
        output_cost = (output_tokens / 1000) * self.rates['output']
        
        self.total_cost += input_cost + output_cost
    
    def get_daily_cost(self):
        return self.total_cost
```

### コスト最適化

#### キャッシュ最適化

```python
class CostOptimizedCache:
    def __init__(self, cost_threshold=0.01):
        self.cache = {}
        self.cost_threshold = cost_threshold
    
    def should_cache(self, estimated_cost):
        return estimated_cost > self.cost_threshold
    
    def get_or_generate(self, prompt, generator, cost_estimator):
        cache_key = hash(prompt)
        
        if cache_key in self.cache:
            return self.cache[cache_key]
        
        estimated_cost = cost_estimator.estimate(prompt)
        
        if not self.should_cache(estimated_cost):
            return generator.generate(prompt)
        
        result = generator.generate(prompt)
        self.cache[cache_key] = result
        return result
```

#### 動的モデル選択

```python
class DynamicModelSelector:
    def __init__(self):
        self.models = {
            'simple': {'model': 'gpt-3.5-turbo', 'cost': 0.002},
            'complex': {'model': 'gpt-4', 'cost': 0.03}
        }
    
    def select_model(self, prompt_complexity):
        if prompt_complexity < 0.5:
            return self.models['simple']['model']
        else:
            return self.models['complex']['model']
    
    def estimate_complexity(self, prompt):
        # 簡易的な複雑度評価
        word_count = len(prompt.split())
        return min(word_count / 100, 1.0)
```

```python
class BatchProcessor:
    def __init__(self, batch_size=10):
        self.batch_size = batch_size
        self.queue = []
    
    def add_request(self, request):
        self.queue.append(request)
        
        if len(self.queue) >= self.batch_size:
            self.process_batch()
    
    def process_batch(self):
        batch = self.queue[:self.batch_size]
```

```python
class PerformanceReview:
    def __init__(self):
        self.baseline_metrics = {}
    
    def set_baseline(self, metrics):
        self.baseline_metrics = metrics
    
    def compare(self, current_metrics):
        improvements = {}
        
        for metric in current_metrics:
            if metric in self.baseline_metrics:
                baseline = self.baseline_metrics[metric]
                current = current_metrics[metric]
                
                if metric == 'latency':
                    improvements[metric] = baseline - current  # 低いほど良い
                else:
                    improvements[metric] = current - baseline
        
        return improvements
```

```python
class ImprovementSuggester:
    def __init__(self):
        self.rules = [
            {'condition': lambda m: m['latency'] > 2, 'suggestion': 'Add caching'},
            {'condition': lambda m: m['error_rate'] > 0.1, 'suggestion': 'Review error handling'},
            {'condition': lambda m: m['cost'] > 1000, 'suggestion': 'Optimize model selection'}
        ]
    
    def suggest(self, metrics):
        suggestions = []
        
        for rule in self.rules:
            if rule['condition'](metrics):
                suggestions.append(rule['suggestion'])
        
        return suggestions
```
